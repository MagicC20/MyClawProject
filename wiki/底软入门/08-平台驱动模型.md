# 08. 平台驱动模型

> 平台驱动是 Linux 嵌入式世界最重要的驱动模型。  
> AI 芯片上的加速器、控制器，几乎都用 platform driver 来实现。

---

## 为什么需要 platform driver

上一章我们写了字符设备驱动，但有个问题：**驱动怎么知道硬件存在？**

两种思路：

1. **老办法**：驱动里 hardcode 一组寄存器地址（编译时就固定）
   ```c
   // 硬编码寄存器地址
   #define MYCHIP_REG_BASE 0xE0000000
   void __iomem *regs = ioremap(MYCHIP_REG_BASE, 0x1000);
   ```
   - 问题：换个板子或换个 SoC 就不能用了
   - 不符合 Linux "硬件无关驱动代码" 的设计哲学

2. **新办法（推荐）**：用 platform bus + 设备树，让驱动和硬件信息解耦
   ```
   设备树描述硬件：寄存器地址、中断号
   platform 驱动匹配硬件 → 加载驱动 → probe() 拿到资源
   ```

platform bus 是 Linux 用来管理 SoC 内部设备的虚拟总线。它不真实存在（不像 USB、PCI），纯粹是为了统一 SoC 上设备的"发现"机制。

---

## platform 总线的三个核心结构

### 1. platform_device：描述"硬件"

```c
// 🔴 内核态
struct platform_device {
    const char *name;            // 设备名
    int id;                      // 设备 id（用于同名多实例）
    struct device dev;           // 通用 device 结构
    u32 num_resources;           // 资源数量
    struct resource *resource;   // 资源数组（寄存器、中断等）
    const struct platform_device_id *id_entry;
    struct pdev_archdata archdata;
};
```

### 2. platform_driver：描述"驱动"

```c
// 🔴 内核态
struct platform_driver {
    int (*probe)(struct platform_device *);   // 设备和驱动匹配时调用
    int (*remove)(struct platform_device *);  // 设备移除时调用
    void (*shutdown)(struct platform_device *);
    int (*suspend)(struct platform_device *, pm_message_t state);
    int (*resume)(struct platform_device *);
    struct device_driver driver;              // 通用 driver 结构
    const struct platform_device_id *id_table;
    const struct of_device_id *of_match_table;  // 设备树匹配表 ← 关键
};
```

### 3. resource：硬件资源

```c
// 🔴 内核态
struct resource {
    resource_size_t start;     // 起始地址或中断号
    resource_size_t end;       // 结束地址（不含）
    const char *name;          // 资源名
    unsigned long flags;       // 类型标志
};

// 类型
#define IORESOURCE_MEM    0x00000200   // 内存区域
#define IORESOURCE_IRQ    0x00000400   // 中断号
```

## platform_device 和 platform_driver 如何匹配

匹配过程（简化版）：

```
platform_bus 启动
   ↓
platform_driver 注册
   ↓
of_match_table 里有 "vendor,device-name"
   ↓
设备树里 .compatible = "vendor,device-name"
   ↓
匹配成功 → 调用 platform_driver.probe()
```

匹配的三种方式：

| 方式 | 适用 | 说明 |
|---|---|---|
| 设备树（of_match_table） | 推荐 | 现代主流，硬件信息在 .dts 文件 |
| ACPI 表 | x86 平台 | 类似设备树，但格式不同 |
| name + id | 老办法 | 不推荐，仅用于遗留代码 |

## 完整示例：platform 驱动

### 设备树描述（mychip.dts）

```dts
// 📄 设备树片段
mychip@0xe0000000 {
    compatible = "vendor,mychip";        // ← 匹配 of_match_table
    reg = <0xe0000000 0x1000>;           // 寄存器：基址 0xe0000000，大小 0x1000
    interrupts = <0 32 4>;               // 中断：SPI 32，触发方式 4
    clocks = <&clock 5>;
    status = "okay";
};
```

### 📄 mychip_drv.c

```c
// 🔴 内核态
#include <linux/module.h>
#include <linux/platform_device.h>
#include <linux/of.h>
#include <linux/of_address.h>
#include <linux/of_irq.h>
#include <linux/io.h>
#include <linux/interrupt.h>
#include <linux/cdev.h>
#include <linux/fs.h>
#include <linux/uaccess.h>

#define DEVICE_NAME "mychip"
#define MYCHIP_MAGIC 'm'
#define MYCHIP_GET_VERSION _IOR(MYCHIP_MAGIC, 0x01, int)
#define MYCHIP_MAX_NR 0x01

struct mychip_priv {
    void __iomem *regs;       // MMIO 寄存器基址
    int irq;                  // 中断号
    struct cdev cdev;
    dev_t dev_num;
    struct class *class;
    struct device *device;
    int version;
};

// --- file_operations ---
static int mychip_open(struct inode *inode, struct file *filp)
{
    struct mychip_priv *priv = container_of(inode->i_cdev,
                                            struct mychip_priv, cdev);
    filp->private_data = priv;
    return 0;
}

static long mychip_unlocked_ioctl(struct file *filp,
                                  unsigned int cmd, unsigned long arg)
{
    struct mychip_priv *priv = filp->private_data;
    int val;

    if (_IOC_TYPE(cmd) != MYCHIP_MAGIC) return -ENOTTY;
    if (_IOC_NR(cmd) > MYCHIP_MAX_NR)   return -ENOTTY;

    switch (cmd) {
    case MYCHIP_GET_VERSION:
        val = priv->version;
        return copy_to_user((int __user *)arg, &val, sizeof(val)) ? -EFAULT : 0;
    default:
        return -ENOTTY;
    }
}

static const struct file_operations mychip_fops = {
    .owner = THIS_MODULE,
    .open  = mychip_open,
    .unlocked_ioctl = mychip_unlocked_ioctl,
};

// --- probe：设备和驱动匹配时调用 ---
static int mychip_probe(struct platform_device *pdev)
{
    struct mychip_priv *priv;
    int ret;

    pr_info("mychip: probing...\n");

    // 1. 分配私有数据结构
    priv = devm_kzalloc(&pdev->dev, sizeof(*priv), GFP_KERNEL);
    if (!priv)
        return -ENOMEM;

    // 2. 获取寄存器资源
    priv->regs = devm_platform_ioremap_resource(pdev, 0);
    if (IS_ERR(priv->regs))
        return PTR_ERR(priv->regs);

    // 3. 获取中断号
    priv->irq = platform_get_irq(pdev, 0);
    if (priv->irq < 0)
        return priv->irq;

    pr_info("mychip: regs=%p irq=%d\n", priv->regs, priv->irq);

    // 4. （可选）注册字符设备
    ret = alloc_chrdev_region(&priv->dev_num, 0, 1, DEVICE_NAME);
    if (ret < 0) return ret;
    cdev_init(&priv->cdev, &mychip_fops);
    priv->cdev.owner = THIS_MODULE;
    ret = cdev_add(&priv->cdev, priv->dev_num, 1);
    if (ret) goto err_chrdev;
    priv->class = class_create(THIS_MODULE, DEVICE_NAME);
    if (IS_ERR(priv->class)) { ret = PTR_ERR(priv->class); goto err_cdev; }
    priv->device = device_create(priv->class, NULL, priv->dev_num,
                                 NULL, DEVICE_NAME);
    if (IS_ERR(priv->device)) { ret = PTR_ERR(priv->device); goto err_class; }

    priv->version = 0x10000;

    // 5. 保存到 pdev->dev
    platform_set_drvdata(pdev, priv);

    pr_info("mychip: probe OK\n");
    return 0;

err_class:
    class_destroy(priv->class);
err_cdev:
    cdev_del(&priv->cdev);
err_chrdev:
    unregister_chrdev_region(priv->dev_num, 1);
    return ret;
}

// --- remove：设备移除时调用 ---
static int mychip_remove(struct platform_device *pdev)
{
    struct mychip_priv *priv = platform_get_drvdata(pdev);

    pr_info("mychip: removing...\n");

    device_destroy(priv->class, priv->dev_num);
    class_destroy(priv->class);
    cdev_del(&priv->cdev);
    unregister_chrdev_region(priv->dev_num, 1);

    return 0;
}

// --- 设备树匹配表 ---
static const struct of_device_id mychip_of_match[] = {
    { .compatible = "vendor,mychip" },
    { /* sentinel */ }
};
MODULE_DEVICE_TABLE(of, mychip_of_match);

// --- platform_driver 结构 ---
static struct platform_driver mychip_driver = {
    .probe  = mychip_probe,
    .remove = mychip_remove,
    .driver = {
        .name = "mychip",
        .of_match_table = mychip_of_match,
    },
};

module_platform_driver(mychip_driver);

MODULE_LICENSE("GPL");
MODULE_AUTHOR("Your Name");
MODULE_DESCRIPTION("Platform driver for mychip");
```

## 关键 API 详解

### `devm_kzalloc` — 带自动释放的内存分配

```c
struct mychip_priv *priv = devm_kzalloc(&pdev->dev, sizeof(*priv), GFP_KERNEL);
```

`devm_*` 系列的"devres"机制：
- 在 probe 中分配
- 在 remove 时**自动释放**
- 不需要在 remove 里手动 kfree

如果中途出错想释放：
```c
priv = devm_kzalloc(...);
if (!priv) return -ENOMEM;

// 错误路径直接 return，devm 会处理
```

### `devm_platform_ioremap_resource` — 获取 MMIO 寄存器

```c
priv->regs = devm_platform_ioremap_resource(pdev, 0);
```

这一行做了三件事：
1. 读设备树里 `reg` 属性
2. 调用 `request_mem_region` 申请 MMIO 区域
3. 调用 `ioremap` 把物理地址映射到内核虚拟地址

返回值是 `void __iomem *`，访问需要用专门的 API：
```c
// 🔴 内核态
iowrite32(value, priv->regs + 0x10);      // 写 32 位寄存器
val = ioread32(priv->regs + 0x14);        // 读 32 位寄存器
u16 v = ioread16(priv->regs + 0x18);      // 读 16 位
u8  b = ioread8(priv->regs + 0x20);       // 读 8 位
```

⚠️ **不要用普通指针解引用 `*priv->regs`！** 必须用 ioread/iowrite，原因：
- 防止编译器优化（I/O 访问不能缓存到寄存器）
- 跨架构兼容（不同 CPU 字节序处理可能不同）

### `platform_get_irq` — 获取中断号

```c
priv->irq = platform_get_irq(pdev, 0);
```

从设备树 `interrupts` 属性读中断号。

⚠️ 新内核（>= 4.4）推荐用 `platform_get_irq` 而不是过时的 `irq_of_parse_and_map`。

### `platform_set_drvdata` 和 `platform_get_drvdata`

```c
platform_set_drvdata(pdev, priv);   // 保存
priv = platform_get_drvdata(pdev);  // 取出
```

用来在 probe 和 remove 之间传递私有数据（因为它们的参数都是 `platform_device *`）。

## 一个驱动多个实例

如果芯片上有 4 个相同 IP，可以在设备树里写：

```dts
mychip_0: mychip@0xe0000000 {
    compatible = "vendor,mychip";
    reg = <0xe0000000 0x1000>;
    interrupts = <0 32 4>;
};
mychip_1: mychip@0xe0001000 {
    compatible = "vendor,mychip";
    reg = <0xe0001000 0x1000>;
    interrupts = <0 33 4>;
};
```

驱动会自动 probe 两次（每个设备一次），每次用不同 `pdev`。`priv` 是独立的。

## 老办法：不用设备树

如果项目还在用老式的 board file（arch/arm/mach-xxx/board-xxx.c）：

```c
// 静态定义 platform_device
static struct resource mychip_resources[] = {
    [0] = {
        .start = 0xE0000000,
        .end   = 0xE0000FFF,
        .flags = IORESOURCE_MEM,
    },
    [1] = {
        .start = 32,
        .end   = 32,
        .flags = IORESOURCE_IRQ,
    },
};

static struct platform_device mychip_device = {
    .name = "mychip",
    .id   = 0,
    .resource = mychip_resources,
    .num_resources = ARRAY_SIZE(mychip_resources),
};

// board_init 里注册
platform_device_register(&mychip_device);
```

⚠️ **新代码不要再用这种方式**。设备树是正确做法。
