# 10. 中断处理

> 中断是驱动的"心跳"。  
> AI 芯片的每一次计算完成都是通过中断通知 CPU 的。

---

## 中断是什么

```
CPU 在执行指令 A
   ↓
硬件产生中断信号（电平变化）
   ↓
CPU 保存现场（PC、寄存器等）
   ↓
跳到中断处理函数（Interrupt Service Routine）
   ↓
处理完返回，继续执行 A
```

中断的本质：硬件主动告诉 CPU "有事发生了"。

应用软件很少接触中断（最多用 signal），但底软几乎每个驱动都要写 ISR（中断服务程序）。

---

## 中断处理的两个核心约束

### 1. 快

ISR 打断了 CPU 正常工作。ISR 跑得越久，系统越卡。

经验：
- 简单 ISR（几十行）：< 10μs
- 复杂 ISR：必须拆分

### 2. 不能睡

ISR 在中断上下文运行，**没有进程可调度**。任何睡眠操作都会死锁。

不能在 ISR 里做的事：
- `kmalloc(GFP_KERNEL)` — 用 `GFP_ATOMIC`
- `mutex_lock` — 用 `spinlock`
- `copy_from_user` — 用户进程在 ISR 时不一定存在
- `printk`（虽然能用，但慢）— 用 `pr_info` 也不要在 ISR 里大量打印

---

## 顶半部 vs 底半部

因为 ISR 必须快，但实际工作往往复杂，所以拆成两部分：

```
硬件中断
   ↓
顶半部（Top Half / Hard IRQ）
   - 极短，只做最关键的事
   - 例如：读状态寄存器、清中断、唤醒底半部
   ↓
底半部（Bottom Half / Soft IRQ）
   - 可以复杂一些
   - 在"软中断"上下文跑，可以睡眠（workqueue 形式）
   - 例如：拷贝数据、通知用户、唤醒等待队列
```

底半部的三种形式：

| 形式 | 上下文 | 能否睡眠 | 适用场景 |
|---|---|---|---|
| Tasklet | 软中断 | 否 | 简单的延迟工作（老 API，逐步废弃）|
| Softirq | 软中断 | 否 | 极高性能需求（网络、块设备用）|
| Workqueue | 进程 | **是** | 复杂工作，需要睡眠 |

**新手推荐：workqueue**（最简单、最安全）。

## request_irq / free_irq

### 注册中断处理函数

```c
// 🔴 内核态
#include <linux/interrupt.h>

typedef irqreturn_t (*irq_handler_t)(int irq, void *dev_id);

// 注册
int request_irq(unsigned int irq,
                irq_handler_t handler,
                unsigned long flags,
                const char *name,
                void *dev_id);

// 释放
void free_irq(unsigned int irq, void *dev_id);
```

### flags 常用值

```c
#define IRQF_SHARED        0x00000080   // 中断共享（多个设备共用一根 IRQ 线）
#define IRQF_TRIGGER_RISING  0x00000001   // 上升沿触发
#define IRQF_TRIGGER_FALLING 0x00000002   // 下降沿触发
#define IRQF_TRIGGER_HIGH    0x00000004   // 高电平触发
#define IRQF_TRIGGER_LOW     0x00000008   // 低电平触发
#define IRQF_ONESHOT         0x00002000   // 线程化中断的一次性标志
```

实际中通常这样组合：
```c
flags = IRQF_SHARED | IRQF_TRIGGER_RISING;
```

### dev_id 的作用

```c
request_irq(irq, handler, flags, "mychip", priv);  // 传 priv 指针
```

`dev_id` 的用途：
1. **共享中断时区分设备**：同一根 IRQ 线多个设备注册，每个 `dev_id` 不同
2. **free_irq 时识别**：传入相同的 `dev_id` 才能正确释放
3. **handler 里取私有数据**：`dev_id` 就是 `priv`

## 完整的 ISR + Workqueue 示例

```c
// 🔴 内核态
#include <linux/interrupt.h>
#include <linux/workqueue.h>

struct mychip_priv {
    void __iomem *regs;
    int irq;
    int event_count;            // 事件计数
    wait_queue_head_t wq;       // 等待队列
    struct work_struct work;    // workqueue 工作项
    spinlock_t lock;
};

static void mychip_work_handler(struct work_struct *work)
{
    struct mychip_priv *priv = container_of(work, struct mychip_priv, work);
    unsigned long flags;

    // 这里是"底半部"，在进程上下文，可以睡眠
    pr_info("mychip: processing event %d\n", priv->event_count);

    // 模拟耗时操作
    msleep(10);

    spin_lock_irqsave(&priv->lock, flags);
    priv->event_count++;
    spin_unlock_irqrestore(&priv->lock, flags);

    wake_up_interruptible(&priv->wq);
}

// ISR：顶半部（极短）
static irqreturn_t mychip_isr(int irq, void *dev_id)
{
    struct mychip_priv *priv = dev_id;
    u32 status;

    // 1. 读状态寄存器
    status = ioread32(priv->regs + 0x100);

    // 2. 检查是不是我们的中断（共享中断需要）
    if (!(status & 0x1))
        return IRQ_NONE;   // 不是，返回 IRQ_NONE 让内核找下一个 handler

    // 3. 清中断标志（写 1 清零）
    iowrite32(status, priv->regs + 0x104);

    // 4. 调度底半部
    schedule_work(&priv->work);

    return IRQ_HANDLED;
}

// probe 里注册
static int mychip_probe(struct platform_device *pdev)
{
    struct mychip_priv *priv;
    int ret;

    // ... 前面获取 regs、irq 的代码 ...

    // 初始化 workqueue
    INIT_WORK(&priv->work, mychip_work_handler);
    init_waitqueue_head(&priv->wq);
    spin_lock_init(&priv->lock);

    // 注册中断
    ret = request_irq(priv->irq, mychip_isr,
                      IRQF_SHARED | IRQF_TRIGGER_HIGH,
                      "mychip", priv);
    if (ret) {
        dev_err(&pdev->dev, "request_irq failed: %d\n", ret);
        return ret;
    }

    return 0;
}

// remove 里释放
static int mychip_remove(struct platform_device *pdev)
{
    struct mychip_priv *priv = platform_get_drvdata(pdev);

    free_irq(priv->irq, priv);
    cancel_work_sync(&priv->work);   // 确保 work 不再运行

    return 0;
}
```

## ISR 返回值

```c
typedef enum irqreturn {
    IRQ_NONE    = (0 << 0),  // 不是我们的中断
    IRQ_HANDLED = (1 << 0),  // 处理了
    IRQ_WAKE_THREAD = (1 << 1),  // 唤醒线程化中断（见下）
} irqreturn_t;
```

- `IRQ_HANDLED`：处理成功
- `IRQ_NONE`：不是我们的中断（共享中断时重要）

## 线程化中断（Threaded IRQ）

有些中断处理需要睡眠操作，又不想用 workqueue 那么重 —— 用线程化中断。

```c
// 🔴 内核态
static irqreturn_t mychip_hard_isr(int irq, void *dev_id)
{
    // 顶半部（硬中断上下文，极短）
    struct mychip_priv *priv = dev_id;
    u32 status = ioread32(priv->regs + 0x100);

    if (status & 0x1) {
        iowrite32(status, priv->regs + 0x104);
        return IRQ_WAKE_THREAD;   // 告诉内核：唤醒线程化的 handler
    }
    return IRQ_NONE;
}

static irqreturn_t mychip_thread_fn(int irq, void *dev_id)
{
    // 底半部（线程上下文，可以睡眠！）
    struct mychip_priv *priv = dev_id;
    pr_info("mychip: thread handler running, can sleep\n");
    msleep(10);
    return IRQ_HANDLED;
}

// 注册时声明 thread_fn
request_threaded_irq(priv->irq,
                     mychip_hard_isr,    // 顶半部
                     mychip_thread_fn,   // 底半部
                     IRQF_ONESHOT,       // ONESHOT 防止重复触发
                     "mychip", priv);
```

线程化中断的好处：
- 顶半部极短（快）
- 底半部可以睡眠（灵活）
- 比 workqueue 更省内存

## 关闭/开启中断

```c
// 🔴 内核态
disable_irq(unsigned int irq);          // 同步关闭（等当前 ISR 跑完）
disable_irq_nosync(unsigned int irq);   // 异步关闭（不等）
enable_irq(unsigned int irq);           // 开启

// 中断屏蔽（当前 CPU 上）
local_irq_disable();
local_irq_enable();
local_irq_save(flags);
local_irq_restore(flags);
```

⚠️ `disable_irq` 有死锁风险（如果你在 ISR 里 disable 自己），慎用。

## /proc/interrupts

```bash
$ cat /proc/interrupts
           CPU0       CPU1       CPU2       CPU3
 24:    1234567    234567    345678    456789   GIC  Level     uart0
 32:          0          0          0          0   GIC  Level     mychip
 33:      12345          0          0          0   GIC  Edge      eth0
```

- 第一列：中断号
- 后面每列：每个 CPU 上该中断的触发次数
- 倒数第二列：中断控制器（GIC = ARM Generic Interrupt Controller）
- 倒数第二列：触发方式（Level / Edge）
- 最后一列：注册时传入的 `name`

判断设备是否在产生中断：看对应行的数字是否在增长。

## 中断上下文 vs 进程上下文判断

```c
// 🔴 内核态
in_interrupt();          // true 表示在中断/软中断上下文
in_irq();                // true 表示在硬 ISR
in_softirq();            // true 表示在软中断

// 经常这样写
if (in_interrupt()) {
    // 必须用 GFP_ATOMIC、spin_lock
} else {
    // 可以睡眠、用 mutex
}
```

---

## 共享中断的注意事项

注册 `IRQF_SHARED` 后：
1. ISR 必须能识别"是不是我的中断"（读状态寄存器）
2. 如果不是，必须返回 `IRQ_NONE`，否则内核会当成你处理了
3. 不能在 ISR 里 disable/enable 中断（共享中断 disable 会影响别人）
