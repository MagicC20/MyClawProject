# 05. 字符设备驱动

> 字符设备是 Linux 驱动里最简单的形式，也是入门最自然的选择。  
> AI 芯片的加速器接口、调试接口、状态查询接口，几乎都是字符设备。

---

## 什么是字符设备

Linux 设备分三大类：

| 类型 | 特点 | 例子 |
|---|---|---|
| **字符设备** | 按字节/字符流访问，无缓冲 | 串口、键盘、GPU、加速器 |
| **块设备** | 按块（512B/4KB）访问，有缓存 | 硬盘、SSD、eMMC |
| **网络设备** | 按包访问，通过 socket | 网卡 |

字符设备的特点：
- 可以像文件一样 `open` / `read` / `write` / `close`
- 但**没有文件系统语义**（不能 `ls`、`cd` 等）
- 通过 `/dev` 目录下的设备文件访问

```
/dev/ttyS0     ← 串口
/dev/null      ← 空设备
/dev/zero      ← 零设备
/dev/mychip    ← 你即将写的字符设备
```

## 字符设备的四个核心数据结构

### 1. 设备号（dev_t）

```c
// 设备号 = 主设备号(Major) + 次设备号(Minor)
// 主设备号：标识一类设备（哪个驱动）
// 次设备号：标识同类设备中的具体一个（哪个实例）

typedef u32 dev_t;
#define MINORBITS   20
#define MINORMASK   ((1U << MINORBITS) - 1)
#define MAJOR(dev)   ((unsigned int)((dev) >> MINORBITS))
#define MINOR(dev)   ((unsigned int)((dev) & MINORMASK))
#define MKDEV(ma,mi) (((ma) << MINORBITS) | (mi))

// 例如：MKDEV(240, 0) → 主设备号 240，次设备号 0
```

主设备号的分配：
- 0~255：传统静态分配（已用完）
- 动态分配：用 `alloc_chrdev_region` 让内核分配一个未用的

### 2. struct cdev：内核里的字符设备对象

```c
// 🔴 内核态
#include <linux/cdev.h>

struct cdev {
    struct kobject kobj;           // 内嵌的 kobject（用于 sysfs）
    struct module *owner;          // 通常是 THIS_MODULE
    const struct file_operations *ops;  // ← 关键：操作函数集
    struct list_head list;         // 链表节点
    dev_t dev;                     // 设备号
    unsigned int count;            // 次设备号个数
    ...
};
```

### 3. struct file_operations：操作函数集

```c
// 🔴 内核态
struct file_operations {
    struct module *owner;
    loff_t (*llseek)(struct file *, loff_t, int);
    ssize_t (*read)(struct file *, char __user *, size_t, loff_t *);
    ssize_t (*write)(struct file *, const char __user *, size_t, loff_t *);
    ssize_t (*read_iter)(struct kiocb *, struct iov_iter *);
    ssize_t (*write_iter)(struct kiocb *, struct iov_iter *);
    int (*iterate)(struct file *, struct dir_context *);
    int (*poll)(struct file *, struct poll_table_struct *);
    long (*unlocked_ioctl)(struct file *, unsigned int, unsigned long);
    long (*compat_ioctl)(struct file *, unsigned int, unsigned long);
    int (*mmap)(struct file *, struct vm_area_struct *);
    int (*open)(struct inode *, struct file *);
    int (*flush)(struct file *, fl_owner_t id);
    int (*release)(struct inode *, struct file *);
    int (*fsync)(struct file *, loff_t, loff_t, int datasync);
    int (*fasync)(int, struct file *, int);
    int (*lock)(struct file *, int, struct file_lock *);
    ...
};
```

新手只需要关心：
- `open` / `release` — 打开/关闭
- `read` / `write` — 数据传输
- `unlocked_ioctl` — 自定义命令
- `mmap` — 内存映射

### 4. struct inode 和 struct file

```c
struct inode {
    // 文件的元数据：权限、大小、设备号...
    dev_t i_rdev;       // 设备文件对应的设备号
    const struct file_operations *i_fop;
    ...
};

struct file {
    // 打开的文件实例（每次 open 都创建一个）
    const struct file_operations *f_op;
    void *private_data; // ← 你可以用这个保存私有数据
    ...
};
```

**关键区别**：
- `inode` 一个文件只有一个（/dev/mychip）
- `file` 每次 `open` 都新建一个（同一文件可以被多次打开）

## 完整示例：mychardev 字符设备

### 📄 mychardev.c

```c
// 🔴 内核态
#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/slab.h>
#include <linux/uaccess.h>

#define DEVICE_NAME "mychardev"
#define BUF_SIZE    4096

static dev_t dev_num;          // 设备号
static struct cdev my_cdev;    // 字符设备对象
static struct class *my_class; // device class（用于自动创建设备文件）
static char *kernel_buf;       // 内核缓冲区

/* === file_operations 实现 === */

static int mychardev_open(struct inode *inode, struct file *filp)
{
    pr_info("mychardev: open() called\n");
    // 可以在这里做设备特定的初始化
    return 0;
}

static int mychardev_release(struct inode *inode, struct file *filp)
{
    pr_info("mychardev: release() called\n");
    return 0;
}

static ssize_t mychardev_read(struct file *filp, char __user *buf,
                              size_t count, loff_t *ppos)
{
    int ret;

    if (*ppos >= BUF_SIZE)
        return 0;  // EOF

    if (count > BUF_SIZE - *ppos)
        count = BUF_SIZE - *ppos;

    ret = copy_to_user(buf, kernel_buf + *ppos, count);
    if (ret)
        return -EFAULT;

    *ppos += count;
    pr_info("mychardev: read %zu bytes\n", count);
    return count;
}

static ssize_t mychardev_write(struct file *filp, const char __user *buf,
                               size_t count, loff_t *ppos)
{
    int ret;

    if (*ppos >= BUF_SIZE)
        return -ENOSPC;

    if (count > BUF_SIZE - *ppos)
        count = BUF_SIZE - *ppos;

    ret = copy_from_user(kernel_buf + *ppos, buf, count);
    if (ret)
        return -EFAULT;

    *ppos += count;
    pr_info("mychardev: write %zu bytes\n", count);
    return count;
}

static const struct file_operations mychardev_fops = {
    .owner   = THIS_MODULE,
    .open    = mychardev_open,
    .release = mychardev_release,
    .read    = mychardev_read,
    .write   = mychardev_write,
};

/* === 模块初始化 === */

static int __init mychardev_init(void)
{
    int ret;

    // 1. 分配设备号（让内核自动选主设备号）
    ret = alloc_chrdev_region(&dev_num, 0, 1, DEVICE_NAME);
    if (ret < 0) {
        pr_err("alloc_chrdev_region failed: %d\n", ret);
        return ret;
    }
    pr_info("mychardev: allocated major=%d, minor=%d\n",
            MAJOR(dev_num), MINOR(dev_num));

    // 2. 初始化 cdev 并添加
    cdev_init(&my_cdev, &mychardev_fops);
    my_cdev.owner = THIS_MODULE;
    ret = cdev_add(&my_cdev, dev_num, 1);
    if (ret < 0) {
        unregister_chrdev_region(dev_num, 1);
        return ret;
    }

    // 3. 创建设备类（让 udev 自动创建设备文件）
    my_class = class_create(THIS_MODULE, DEVICE_NAME);
    if (IS_ERR(my_class)) {
        cdev_del(&my_cdev);
        unregister_chrdev_region(dev_num, 1);
        return PTR_ERR(my_class);
    }

    // 4. 创建设备文件 /dev/mychardev
    device_create(my_class, NULL, dev_num, NULL, DEVICE_NAME);
    pr_info("mychardev: device /dev/%s created\n", DEVICE_NAME);

    // 5. 分配内核缓冲区
    kernel_buf = kzalloc(BUF_SIZE, GFP_KERNEL);
    if (!kernel_buf) {
        device_destroy(my_class, dev_num);
        class_destroy(my_class);
        cdev_del(&my_cdev);
        unregister_chrdev_region(dev_num, 1);
        return -ENOMEM;
    }

    return 0;
}

static void __exit mychardev_exit(void)
{
    kfree(kernel_buf);
    device_destroy(my_class, dev_num);
    class_destroy(my_class);
    cdev_del(&my_cdev);
    unregister_chrdev_region(dev_num, 1);
    pr_info("mychardev: unloaded\n");
}

module_init(mychardev_init);
module_exit(mychardev_exit);

MODULE_LICENSE("GPL");
MODULE_AUTHOR("Your Name");
MODULE_DESCRIPTION("A simple character device driver");
```

## 关键流程图

```
insmod mychardev.ko
   ↓
alloc_chrdev_region(&dev_num, ...)    ← 内核分配主设备号
   ↓
cdev_init(&my_cdev, &fops)            ← 把 file_operations 绑到 cdev
   ↓
cdev_add(&my_cdev, dev_num, 1)        ← 把 cdev 加入内核
   ↓
class_create(...)                      ← 创建 /sys/class/mychardev
   ↓
device_create(...)                    ← 创建 /dev/mychardev
   ↓
用户 open("/dev/mychardev")
   ↓
mychardev_open()                      ← 你的 open 实现
   ↓
用户 read()/write()/ioctl()
   ↓
mychardev_read()/write()/ioctl()      ← 你的实现
   ↓
用户 close()
   ↓
mychardev_release()                   ← 你的 release 实现
```

## 常见问题

### 1. 设备号冲突 / 加载失败

```bash
$ sudo insmod mychardev.ko
mychardev: alloc_chrdev_region failed: -16
                                          ↑
                                       EBUSY

# 解决：用动态分配（alloc_chrdev_region）而不是静态指定
```

### 2. /dev/mychardev 没有自动创建

可能是 udev 没启动或 class_create 失败。检查内核日志：
```bash
$ sudo dmesg | tail -10
```

老内核（< 3.10）需要手动 `mknod`：
```bash
$ sudo mknod /dev/mychardev c 236 0   # c=字符设备, 236=主, 0=次
```

### 3. open 报 `Permission denied`

设备文件权限太严：
```bash
$ sudo chmod 666 /dev/mychardev
```

## minor = 多个次设备

如果一个驱动要管多个相同设备（如 4 个串口），可以让 minor 区分：

```c
// 分配 4 个次设备（0, 1, 2, 3）
alloc_chrdev_region(&dev_num, 0, 4, "myuart");

// 注册 4 个 cdev（或一个 cdev + count=4）
my_cdev.count = 4;
cdev_add(&my_cdev, dev_num, 4);

// 在 open 中区分
static int myuart_open(struct inode *inode, struct file *filp)
{
    int minor = MINOR(inode->i_rdev);
    pr_info("opening uart %d\n", minor);
    filp->private_data = &uart_devices[minor];
    return 0;
}
```

## 本章小结

✅ **本章速查表**

| 元素 | 用途 |
|---|---|
| `dev_t` | 设备号 = 主设备号 + 次设备号 |
| `alloc_chrdev_region` | 动态分配设备号 |
| `unregister_chrdev_region` | 释放设备号 |
| `struct cdev` | 内核里的字符设备对象 |
| `cdev_init` / `cdev_add` / `cdev_del` | cdev 的初始化/添加/删除 |
| `struct file_operations` | 操作函数集（open/read/write/ioctl...）|
| `class_create` / `class_destroy` | 创建设备类（/sys/class）|
| `device_create` / `device_destroy` | 创建设备文件（/dev/...）|
| `MAJOR` / `MINOR` / `MKDEV` | 设备号分解/合并 |
